async function loadHistory() { const latest = await provider.getBlockNumber(); const rows = []; let to = latest; for (let chunk = 0; chunk < 10 && to > 0; chunk++) { const from = Math.max(0, to - 9999); const [created, assigned, completed, fired] = await Promise.all([ vault.queryFilter(vault.filters.TaskCreated(), from, to), vault.queryFilter(vault.filters.AgentAssigned(), from, to), vault.queryFilter(vault.filters.TaskCompleted(), from, to), vault.queryFilter(vault.filters.FirewallTriggered(), from, to), ]); rows.push( ...created.map(e => ({ b: e.blockNumber, type: "ESCROW", text: `Task #${e.args.taskId} escrowed — ${bnb(e.args.reward)} locked by ${short(e.args.client)}` })), ...assigned.map(e => ({ b: e.blockNumber, type: "ASSIGN", text: `Agent ${short(e.args.agent)} staked ${bnb(e.args.stakeLocked)} on task #${e.args.taskId}` })), ...completed.map(e => ({ b: e.blockNumber, type: "COMPLETE",text: `Task #${e.args.taskId} settled — ${bnb(e.args.rewardPaid)} released` })), ...fired.map(e => ({ b: e.blockNumber, type: "FIREWALL", text: `KILL SWITCH on task #${e.args.taskId} — ${bnb(e.args.slashedAmount)} slashed. Reason: ${e.args.reason}` })) ); if (created.some(e => Number(e.args.taskId) === 1)) break; // reached the very first task to = from - 1; } rows.sort((x, y) => x.b - y.b).forEach(r => addRow(r.type, r.text)); }
